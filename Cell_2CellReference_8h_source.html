23// Fast base-26 digit lookup: maps 'A'-'Z' and 'a'-'z' to 1-26, 0 for invalid.
24inline constexpr unsigned char base26Lookup(unsigned char c) noexcept {
25 if (c >= 'A' && c <= 'Z') return static_cast<unsigned char>(c - 'A' + 1);
26 if (c >= 'a' && c <= 'z') return static_cast<unsigned char>(c - 'a' + 1);
27 return 0;
28}

31// Compact cell key: (row << 20) | column. Max row 1,048,576 (20 bits), max col 16,384 (14 bits).
