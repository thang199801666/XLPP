31// Conditional format value object (cfvo): the min/max/midpoint of a data bar,
32// color scale, or icon set threshold.
33struct Cfvo {
34 // "num", "percent", "percentile", "min", "max", "formula"
35 std::string type{"num"};
36 double value{0};
37 bool hasValue{false};
38 std::string formula;
39 std::optional<std::string> color; // ARGB for color scale stops
40
41 Cfvo() = default;
42 explicit Cfvo(std::string type_, double val) : type(std::move(type_)), value(val), hasValue(true) {}
43 Cfvo(std::string type_, std::string formula_) : type(std::move(type_)), formula(std::move(formula_)), hasValue(true) {}
44};
