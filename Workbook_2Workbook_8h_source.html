19// A package part that XLPP does not model (custom XML, charts, VBA, ...).
20// `load` captures such parts verbatim so a subsequent `save` round-trips them
21// instead of silently dropping them.
22struct PreservedPart {
23 std::string name; // package path, e.g. "customXml/item1.xml"
24 std::string data; // raw part bytes
25 std::string overrideType; // content type to emit as <Override> ("" if covered by a <Default>)
26 std::string extension; // file extension ("" if none)
27 std::string defaultType; // content type of the <Default> rule ("" if none)
28 bool compress{true}; // whether to compress this part on save
29};
