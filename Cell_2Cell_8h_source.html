17// Excel 365 dynamic array / new functions that require the _xlfn. prefix in
18// the stored formula so Excel 2016-2019 & 365 parse them correctly. Returns the
19// prefixed form, or the input unchanged if it already has a prefix.
20inline std::string xlfn(std::string_view function) {
21 if (function.empty()) return {};
22 if (function.size() > 6 && function.substr(0, 6) == "_xlfn.") return std::string(function);
23 return "_xlfn." + std::string(function);
24}
