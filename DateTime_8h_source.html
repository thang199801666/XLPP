25// Days since 1970-01-01 for a civil (proleptic Gregorian) date. Port of the
26// well-known days_from_civil algorithm.
27inline long long daysFromCivil(int year, int month, int day) noexcept {
28 year -= month <= 2;
29 const long long era = (year >= 0 ? year : year - 399) / 400;
30 const unsigned yearOfEra = static_cast<unsigned>(year - era * 400);
31 const unsigned dayOfYear = (153 * (month + (month > 2 ? -3 : 9)) + 2) / 5 + day - 1;
32 const unsigned dayOfEra = yearOfEra * 365 + yearOfEra / 4 - yearOfEra / 100 + dayOfYear;
33 return era * 146097 + static_cast<long long>(dayOfEra) - 719468;
34}

36inline void civilFromDays(long long days, int& year, int& month, int& day) noexcept {
37 days += 719468;
38 const long long era = (days >= 0 ? days : days - 146096) / 146097;
39 const unsigned dayOfEra = static_cast<unsigned>(days - era * 146097);
40 const unsigned yearOfEra = (dayOfEra - dayOfEra / 1460 + dayOfEra / 36524 - dayOfEra / 146096) / 365;
41 const long long yearOfCycle = static_cast<long long>(yearOfEra) + era * 400;
42 const unsigned dayOfYear = dayOfEra - (365 * yearOfEra + yearOfEra / 4 - yearOfEra / 100);
43 const unsigned monthPrime = (5 * dayOfYear + 2) / 153;
44 day = static_cast<int>(dayOfYear - (153 * monthPrime + 2) / 5 + 1);
45 month = static_cast<int>(monthPrime + (monthPrime < 10 ? 3 : -9));
46 year = static_cast<int>(yearOfCycle + (month <= 2));
47}

49// Converts a calendar date/time to the Excel serial day number. The 1900 system
50// counts from 1899-12-31 with a phantom 1900-02-29 (serial 60), matching Excel's
51// leap-year bug; the 1904 system counts from 1904-01-01 with no such bug.
52inline double toExcelSerial(const DateTime& value, bool date1904 = false) noexcept {
53 const long long days = daysFromCivil(value.year, value.month, value.day);
54 const long long epoch = date1904 ? daysFromCivil(1904, 1, 1) : daysFromCivil(1899, 12, 31);
55 long long serial = days - epoch;
56 if (!date1904 && serial >= 60) ++serial;
57 const double fraction = (value.hour * 3600.0 + value.minute * 60.0 + value.second) / 86400.0;
58 return static_cast<double>(serial) + fraction;
59}

61inline DateTime fromExcelSerial(double serial, bool date1904 = false) noexcept {
62 const long long whole = static_cast<long long>(std::floor(serial));
63 const double fraction = serial - static_cast<double>(whole);
64 long long days = whole;
65 if (!date1904 && days >= 60) --days;
66 const long long epoch = date1904 ? daysFromCivil(1904, 1, 1) : daysFromCivil(1899, 12, 31);
67 DateTime result;
68 civilFromDays(epoch + days, result.year, result.month, result.day);
69 double totalSeconds = fraction * 86400.0;
70 totalSeconds = std::round(totalSeconds * 1000.0) / 1000.0;
71 const long long wholeSeconds = static_cast<long long>(totalSeconds);
72 const double subsecond = totalSeconds - static_cast<double>(wholeSeconds);
73 result.hour = static_cast<int>((wholeSeconds / 3600) % 24);
74 result.minute = static_cast<int>((wholeSeconds / 60) % 60);
75 result.second = static_cast<double>(wholeSeconds % 60) + subsecond;
76 return result;
77}

79// True when an Excel number-format code renders values as dates or times, used

83// sections and escaped characters are ignored.
84inline bool isDateFormatCode(std::string_view format, int numFmtId = -1) noexcept {
85 if (numFmtId >= 0) {
86 if (numFmtId == 14 || numFmtId == 15 || numFmtId == 16 || numFmtId == 17 ||
87 numFmtId == 18 || numFmtId == 19 || numFmtId == 20 || numFmtId == 21 ||
88 numFmtId == 22) return true;
89 if (numFmtId >= 27 && numFmtId <= 36) return true;
90 if (numFmtId >= 45 && numFmtId <= 47) return true;
91 if (numFmtId >= 50 && numFmtId <= 58) return true;
92 if (numFmtId == 81) return true;
93 }
94 bool quoted = false;
95 for (std::size_t i = 0; i < format.size(); ++i) {
96 const char c = format[i];
97 if (quoted) {
98 if (c == '"') quoted = false;
99 continue;
100 }
101 if (c == '"') { quoted = true; continue; }
102 if (c == '[') {
103 while (i < format.size() && format[i] != ']') ++i;
104 continue;
105 }
106 if (c == '\\') { ++i; continue; }
107 if (c == 'y' || c == 'm' || c == 'd' || c == 'h' || c == 's') return true;
108 }
109 return false;
110}
